cmake_minimum_required(VERSION 3.16)
project(OmniCompass LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(omnicompass STATIC src/core.cpp src/csv.cpp src/governor.cpp src/shield.cpp src/hpa.cpp)
target_include_directories(omnicompass PUBLIC include)
add_executable(oc_run_fixture tools/run_fixture.cpp)
target_link_libraries(oc_run_fixture PRIVATE omnicompass)
add_executable(oc_governor tools/run_governor.cpp)
target_link_libraries(oc_governor PRIVATE omnicompass)
add_executable(oc_savings tools/savings.cpp)
add_executable(oc_hpa tools/run_hpa.cpp)
target_link_libraries(oc_hpa PRIVATE omnicompass)
add_executable(oc_shield tools/run_shield.cpp)
target_link_libraries(oc_shield PRIVATE omnicompass)
add_executable(oc_soak tools/soak.cpp)
target_link_libraries(oc_soak PRIVATE omnicompass)
add_executable(oc_smoke tools/smoke.cpp)
target_link_libraries(oc_smoke PRIVATE omnicompass)
enable_testing()
add_test(NAME core_smoke COMMAND oc_smoke)
